# Invariants that must hold on any OS and CPU.
# Exact numbers are printed by run.sh into results/actual.txt.

# Live stack trace: runtime async shows the real call chain without builder frames
stack.runtime.frames < stack.classic.frames
stack.classic.builder_frames > 0
stack.runtime.builder_frames == 0

# Synchronously completed two-level chain: runtime async allocates less.
# Not zero in this probe: the outermost boundary still materializes the Task<int>
# that the delegate blocks on. BenchmarkDotNet numbers are in the bench job.
alloc.completed.classic.bytes_per_op > 0
alloc.completed.runtime.bytes_per_op < alloc.completed.classic.bytes_per_op

# Chain with Task.Yield is printed but not checked: blocking on a suspended task
# with GetResult() adds wait allocations that hide the difference between the two models.

# Exception thrown at depth 30: runtime async allocates less
alloc.throw30.runtime.bytes_per_op < alloc.throw30.classic.bytes_per_op

# Ten nested async methods: runtime async produces a smaller assembly
size.runtime.bytes < size.classic.bytes
